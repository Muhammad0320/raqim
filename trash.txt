pub struct AegisGateKeeper {
    // AEGIS ENGINE: Maps an Agent's Hex ID to their specific security policy.
    policies: RwLock<HashMap<String, AegisPolicy>>,

    // TRUE CONCURRENCY: Dashmaps shards the locks. O(1) reads without blocking writers.
    pub quarantine_blocklist: DashMap<String, QuarantineRecord>,
    tx: Sender<SystemEvent>,

    ui_tx: Sender<UiEvent>,
}

impl AegisGateKeeper {
    pub fn new(aegis_path: &str, tx: Sender<SystemEvent>, ui_tx: Sender<UiEvent>) -> Arc<Self> {
        let initial_config = Self::parse_toml(&aegis_path);

        let gatekeeper = Arc::new(Self {
            policies: RwLock::new(initial_config),
            quarantine_blocklist: DashMap::new(),
            tx,
            ui_tx,
        });

        // Spawn a dedicated bg thread for the C-level fs watcher
        let path_string = aegis_path.to_string();
        let gk_clone = gatekeeper.clone();

        // 3. The Async Tokio task that actually swaps the memory.
        std::thread::spawn(move || {
            let (tx, rx) = channel();
            let mut watcher =
                notify::recommended_watcher(tx).expect("Failed to bind os file to watcher");
            watcher
                .watch(
                    std::path::Path::new(&path_string),
                    RecursiveMode::NonRecursive,
                )
                .unwrap();

            // This thread blocks efficiently until the OS sends a file modifiication event.
            for res in rx {
                match res {
                    Ok(event) => {
                        // We only care if the file content were actually modified
                        if let EventKind::Modify(_) = event.kind {
                            println!("[AEGIS] Modification detected. Hot reloadidng ACL...");

                            // Parse the updated file
                            let new_policies = Self::parse_toml(&path_string);

                            //  FAIL-SAFE: Only apply if the new file actually parsed correctly
                            if !new_policies.is_empty() {
                                // Obtain write lock, swap the mappig, instantly release the lock
                                let mut write_lock = gk_clone.policies.write().unwrap();
                                *write_lock = new_policies;
                                println!("[AEGIS] ACL Hot-Reloaded Successfully.")
                            }
                        }
                    }

                    Err(e) => eprintln!("[AEGIS] Watcher Error: {:?}", e),
                }
            }
        });

        gatekeeper
    }


    /// The Semantic Interdiction Switch. Returns TRUE if allowed, FALSE if malicious.
    pub fn enforce_aegis_policy(&self, agent_hex: &str, intent_path: &str) -> bool {
        // Check quarantine first (0(1) instant rejection )
        if self.quarantine_blocklist.contains_key(agent_hex) {
            eprintln!(
                "[AEGIS] PARALYSED AGENT {} ATTEMPTED ACTION. DROPPED. ",
                agent_hex
            );
            return false;
        }

        // 2. Evaluate Semantic Namespace policy.
        let policies = self.policies.read().unwrap();
        if let Some(policy) = policies.get(agent_hex) {
            // Check explicit Blocks (e.g., "raqim_finance/*")
            for blocked in &policy.blocked_namespaces {
                let is_breach = if blocked.ends_with("*") {
                    // Zero-copy slice
                    intent_path.starts_with(&blocked[..blocked.len() - 1])
                } else {
                    intent_path == blocked
                };

                if is_breach {
                    self.trigger_quarantine(
                        agent_hex,
                        intent_path,
                        "NAMESPACE_BREACH",
                        "Attempted execution via unauthorized channel",
                    );
                    return false;
                }
            }

            // Check explicit allows
            for allowed in &policy.allowed_namespaces {
                let is_authorized = if allowed.ends_with("*") {
                    intent_path.starts_with(&allowed[..allowed.len() - 1])
                } else {
                    intent_path == allowed
                };

                if is_authorized {
                    return true;
                }
            }

            // Default Deny if not explicitly allowed
            self.trigger_quarantine(
                agent_hex,
                intent_path,
                "NAMESPACE_BREACH",
                "Attempted execution via unauthorized channel",
            );
            return false;
        }

        //  In strict mode, default is false. For dev it's true
        true
    }


    /// Evaluates if an agent is authorized to communicate with a specific service capability
    pub fn enforce_a2a_policy(&self, sender_hex: &str, target_capability: &str) -> bool {
        if self.quarantine_blocklist.contains_key(sender_hex) {
            return false;
        }

        let policies = self.policies.read().unwrap();

        if let Some(policy) = policies.get(sender_hex) {
            // 1. Checks Explicit Blocks first
            for blocked in &policy.blocked_namespaces {
                if target_capability.starts_with(blocked) {
                    self.trigger_quarantine(
                        sender_hex,
                        target_capability,
                        "NAMESPACE_BREACH",
                        "A2A Blocked Namespace Violation",
                    );
                    return false;
                }
            }

            // Checks if the agents allowed namespace covers the target capability
            for allowed in &policy.allowed_namespaces {
                if target_capability.starts_with(allowed) {
                    return true;
                }
            }
        }

        // Zero-Trust Default Deny
        self.trigger_quarantine(
            sender_hex,
            target_capability,
            "NAMESPACE_BREACH",
            "A2A Unauthorized Capability Access",
        );
        false
    }

    /// Universally verifies any crytographic payload at the network edge
    pub fn verify_agent_signature(
        &self,
        agent_hex: &str,
        payload: &[u8],
        signature_bytes: &[u8; 64],
    ) -> bool {
        let policies = self.policies.read().unwrap();

        if let Some(policy) = policies.get(agent_hex) {
            // Decode the expected public_key from the TOML
            let pub_key_bytes = hex::decode(&policy.public_key_hex).unwrap_or_default();

            if let Ok(public_key) =
                VerifyingKey::from_bytes(pub_key_bytes.as_slice().try_into().unwrap_or(&[0; 32]))
            {
                let signature =
                    Signature::from_bytes(signature_bytes.try_into().unwrap_or(&[0; 64]));

                // Mathematically prove that sender owns this privage key
                if public_key.verify(payload, &signature).is_ok() {
                    return true;
                }
            }
        }

        // If the key is missing, invalid, or signature is spoofed quarantine them.
        self.trigger_quarantine(
            agent_hex,
            "Global",
            "CRYPTO_SPOOF",
            "Invalid Ed25519 Signature.",
        );
        false
    }
}

/// The shared cryptographic token layout compiled across the ecosystem nodes
#[derive(Serialize, Deserialize, Debug, Clone)]
pub struct CapabilityCertificate {
    pub agent_hex: String,
    pub group_name: String,
    pub allowed_namespaces: Vec<String>,
    pub blocked_namespaces: Vec<String>,
    pub expiration_timestamp: u64,
    pub master_signature: Vec<u8>,
}

#[derive(Parser)]
#[command(name = "raqim", about = "Raqim OS Decentralized Administration CLI", version = "2.0")]
struct Cli {

    #[arg(short, long, default_value = "http://127.0.0.1:8081", global = true)]

    /// Enterprise Master License JWT for control plane endpoint authentication

    license_key: Option<String>,

    #[command(subcommand)]
    command: Commands,
}

#[derive(Subcommand)]
enum Commands {
    /// Decentralized Cryptographic Fleet Identity Management
    Keys {
        #[command(subcommand)]
        action: KeyAction,
    },

    /// Aegis Live Firewall Control Matrix Management
    Aegis {
        #[command(subcommand)]
        action: AegisAction,
    },

    /// Time Machine & Reality Forking Administration
    TimeTravel {
        /// Unique 16-byte identifying hex token of the target agent context

        /// Absolute immutable Transaction ID checkpoint to roll back to

        #[arg(short, long)]

#[derive(Subcommand)]
enum KeyAction {
    /// Mints an isolated batch of verifiable agent identities signed by the Master CA Key
    Generate {
        /// Base filename stem prefix for the minted file bundle
        #[arg(short, long)]
        name: String,

        /// The live policy target security group defined inside the server's aegis.toml
        #[arg(short, long)]
        group: String,

        /// Total count of individual key/passport sets to generate concurrently
        #[arg(short, long, default_value_t = 1)]
        count: u32,

        /// File system path pointing to the secret Master Private PEM/DAT Key block
        #[arg(short, long, default_value = "./keys/swarm_master.pem", env = "RAQIM_MASTER_KEY")]
        master_key_path: String,

        /// Absolute path to the secure output storage container directory
        #[arg(short, long, default_value = "./vault/identities")]
        output_dir: String,

        /// Lifetime duration constraint of the certificates in days
        #[arg(short, long, default_value_t = 365)]
        days: u64,
    },
}

#[derive(Subcommand)]
enum AegisAction {
    /// Lists all runtime nodes currently isolated inside the Aegis quarantine map
    List,

    /// Lifts isolation bounds on a quarantined agent, restoring data plane access
    Lift { 
        agent_id: String 
    },
}

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cli = Cli::parse();
    let http_client = Client::builder().build()?;

    let get_auth = || -> String {
        cli.license_key
            .clone()
            .expect("FATAL ERROR: Admin authorization token required. Pass via --license-key or set RAQIM_LICENSE_KEY")
    };

    match &cli.command {
        Commands::Keys {
            action: KeyAction::Generate {
                name,
                group,
                count,
                master_key_path,
                output_dir,
                days,
            },
        } => {
            println!("[CA CLIENT] Initializing out-of-band identity minting pass...");

            // 1. Load the Swarm Master Private CA Signing Key from disk
            if !Path::new(master_key_path).exists() {
                eprintln!("FATAL CA ERROR: Swarm Master Key file not found at path: '{}'", master_key_path);
                std::process::exit(1);
            }
            let master_key_bytes = fs::read(master_key_path)?;
            let master_array: &[u8; 32] = master_key_bytes.as_slice().try_into().map_err(|_| {
                "Cryptographic Identity Failure: Master Key file footprint must be exactly 32 raw bytes long"
            })?;
            let master_signing_key = SigningKey::from_bytes(master_array);

            // 2. Resolve target storage paths
            fs::create_dir_all(output_dir)?;

            // Calculate expiration timestamp bounds
            let seconds_in_day = 86400;
            let expiration_timestamp = SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs() + (days * seconds_in_day);

            let mut csprng = OsRng;
            println!("[CA CLIENT] Stamping certificates for Role Group: '{}'", group);

            // 3. Loop through execution allocations to generate the identities
            for i in 1..=*count {
                let current_signing_key = SigningKey::generate(&mut csprng);
                let current_public_key = current_signing_key.verifying_key();

                // Derive the 16-byte unique identifying token via MD5 hash pass
                let mut hasher = Md5::new();
                hasher.update(current_public_key.to_bytes());
                let agent_id_bytes: [u8; 16] = hasher.finalize().into();
                let agent_hex = hex::encode(agent_id_bytes);

                // 4. Construct the unsigned CapabilityCertificate template frame
                let mut certificate = CapabilityCertificate {
                    agent_hex: agent_hex.clone(),
                    group_name: group.clone(),
                    allowed_namespaces: Vec::new(), // Enforced via server-side hot manifests instead
                    blocked_namespaces: Vec::new(),
                    expiration_timestamp,
                    master_signature: Vec::new(),
                };

                // 5. Generate the authentic signature over the structural bytes
                let serialized_unsigned = postcard::to_allocvec(&certificate)?;
                let master_signature = master_signing_key.sign(&serialized_unsigned);
                certificate.master_signature = master_signature.to_bytes().to_vec();

                // Pack the final token payload using postcard
                let final_certificate_payload = postcard::to_allocvec(&certificate)?;

                // 6. Enforce file naming stems to ensure compatibility with hot reloaders
                let file_suffix = if *count > 1 { format!("{}_{:03}", name, i) } else { name.clone() };
                let key_dest_path = format!("{}/{}.key", output_dir, file_suffix);
                let cert_dest_path = format!("{}/{}.cert", output_dir, file_suffix);

                // Write the components out to disk
                fs::write(&key_dest_path, current_signing_key.to_bytes())?;
                fs::write(&cert_dest_path, final_certificate_payload)?;

                // Unix Security Guard: Restrict private key read permissions to owner
                #[cfg(unix)]
                {
                    use std::os::unix::fs::PermissionsExt;
                    fs::set_permissions(&key_dest_path, fs::Permissions::from_mode(0o600))?;
                }

                println!(
                    "  [+] Minted node [{}/{}]: ID: {} -> Files: {}.key/.cert", 
                    i, count, agent_hex, file_suffix
                );
            }
            println!("✅ Fleet generation completed successfully. Output isolated inside: '{}'", output_dir);
        }

        Commands::Aegis { action: AegisAction::List } => {
            let url = format!("{}/v1/admin/quarantine", cli.daemon_url);
            let res = http_client
                .get(&url)
                .header("Authorization", format!("Bearer {}", get_auth()))
                .send()
                .await?;

            if res.status().is_success() {
                let agents: Vec<String> = res.json().await?;
                println!("🔒 Quarantined Nodes Inside Active Memory: {:?}", agents);
            } else {
                eprintln!("❌ Operational Failure: Control plane returned status: {}", res.status());
            }
        }

        Commands::Aegis { action: AegisAction::Lift { agent_id } } => {
            let url = format!("{}/v1/admin/quarantine/lift", cli.daemon_url);
            let res = http_client
                .post(&url)
                .header("Authorization", format!("Bearer {}", get_auth()))
                .json(&json!({ "agent_id": agent_id }))
                .send()
                .await?;

            if res.status().is_success() {
                println!("🔓 Aegis perimeter updated. Lifted isolation bounds for agent: {}", agent_id);
            } else {
                eprintln!("❌ Operational Failure: Control plane rejected request. Status: {}", res.status());
            }
        }

        Commands::TimeTravel { agent_id, tx_id, fork_config } => {
            let mut payload = json!({
                "agent_id": agent_id,
                "target_tx_id": tx_id,
                "fork_config": null
            });

            if let Some(config_path) = fork_config {
                let config_str = fs::read_to_string(config_path)?;
                let fork_json: serde_json::Value = serde_json::from_str(&config_str)?;
                payload["fork_config"] = fork_json;
            }

            let url = format!("{}/v1/admin/time_travel", cli.daemon_url);
            println!("⌛ Communicating with Time Machine... Forking reality loop for agent {} down to checkpoint TxID: {}...", agent_id, tx_id);

            let res = http_client
                .post(&url)
                .header("Authorization", format!("Bearer {}", get_auth()))
                .json(&payload)
                .send()
                .await?;

            if res.status().is_success() {
                println!("⚡ Reality branch successfully initialized. Isolated simulation task active on core pipeline.");
            } else {
                eprintln!("❌ Time Travel Failure: Control plane rejected execution context. Status: {}", res.status());
            }
        }
    }

    Ok(())
}

    topic_name: String,
    brain: Arc<SwarmStateRegistry>,
    axon: Arc<AxonGateKeeper>,
    tx: Sender<SystemEvent>,
) {

        //  Initialize inside the thread
        let cortex = CortexDataPlane::new(&topic_name);

        let subscriber = cortex
            .create_subscriber()
            .expect("Failed to create local subscriber");

        println!("Cortx Data Plane: Listening for zero-copy local thoughts...");

        //  Read from shared physical RAM
        loop {
            if let Ok(Some(sample)) = subscriber.receive() {
                let payload_bytes = sample.payload();

                // Zero copy deserialization of the dynamic bytes
                let archived_log = unsafe {
                    rkyv::access_unchecked::<<OpLog as rkyv::Archive>::Archived>(payload_bytes)
                };

                // The Circuit Breaker
                if axon.verify_foreign_thoughts(&archived_log) {
                    brain
                        .get_or_create_brain(&archived_log.state.namespace)
                        .assimilate_foreign_thought(&archived_log.delta.as_slice())
                        .expect("FATAL: failed to assimilate thought");
                    println!(
                        "Cortex: Assimilated thought from Agent: {:?}",
                        archived_log.agent_id.as_slice()
                    );

                    println!("CRITICAL ALERT: Local tampering detected. Dropping thoughts.");

                        agent_id: hex::encode(&archived_log.agent_id.as_slice()),
                        reason: "Local tampering detected - Markle Hash Mismatch ".to_string(),
                        culprit_text: archived_log.state.text.as_str().to_string(),
                    });
                }
            }

            std::thread::yield_now();
        }
    });
